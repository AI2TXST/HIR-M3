\section{Supplementary Methods}

\subsection{Prediction-Time Feature Audit and Leakage Elimination}
To guarantee translational clinical validity and strict adherence to the TRIPOD-AI reporting standards, all features were audited against their availability at the index time $t_0$, defined as the completion timestamp of the initial Start of Care (SOC) or Resumption of Care (ROC) OASIS assessment. Variables were categorized into three permissible inductive tiers:
\begin{enumerate}
    \item \textbf{Micro-Level (Individual Clinical \& Demographic State)}: Patient age, sex, self-reported race/ethnicity, body mass index (BMI) continuous score and CDC categorical classifications (Underweight, Normal, Overweight, Obese Classes I--III), baseline clinical disciplines involved (\texttt{ByDiscipline\_RN}, \texttt{PT}, \texttt{OT}, \texttt{SLP}), baseline Charlson Comorbidity Index scores (\texttt{charlson\_score}, \texttt{charlson\_ageadj}, 10-year survival probability), Elixhauser Comorbidity Index scores (\texttt{elix\_quan\_score}, \texttt{elix\_quan\_ageadj}, \texttt{elix\_swiss\_score}), 31 binary Elixhauser condition indicator flags, primary/secondary ICD-10 diagnostic cluster groupings, and 32-dimensional local HICD-BERT diagnostic embeddings.
    \item \textbf{Meso-Level (Community \& Environmental SDOH Context)}: USDA RUCA urban/rural categorization, Census 2020 tract population counts and urban/rural density fractions (\texttt{POP\_URB}, \texttt{POPPCT\_URB}, \texttt{POP\_RUR}, \texttt{POPPCT\_RUR}), and AHRQ/ACS tract-level social determinants of health (tract poverty ratios, educational attainment fractions, household broadband access, computer ownership, vehicle availability, and housing crowding indices).
    \item \textbf{Macro-Level (Institutional \& Care-Delivery Setting)}: Admission Home Health Resource Group HIPPS code (\texttt{Submitted\_HIPPS\_Code}), and out-of-fold target-frequency encodings for facility internal IDs (\texttt{Facility\_Internal\_ID\_freq}) and agency Medicare provider numbers (\texttt{Agency\_Medicare\_Number\_freq}).
\end{enumerate}
All post-index features were strictly eliminated: episode care duration (\texttt{Days\_Cared\_For}, computed as the elapsed days between Start of Care and Discharge Date), post-baseline visit counts (\texttt{NumVisits}), inter-visit dynamics (\texttt{DaysBetweenVisits}), end-of-episode mortality flags (\texttt{ever\_deceased}), and non-generalizable administrative identifiers (\texttt{BENE\_ID}, \texttt{COUNTYFIPS}, patient ZIP codes).

\subsection{Preprocessing and Data Partitioning}
We implemented a two-stage nested partitioning strategy:
$$\mathcal{D}_{\text{Total}} \xrightarrow{80/20} \{\mathcal{D}_{\text{Dev}} \; (80\%), \; \mathcal{D}_{\text{Test}} \; (20\%)\}$$
$$\mathcal{D}_{\text{Dev}} \xrightarrow{80/20} \{\mathcal{D}_{\text{Train}} \; (64\%), \; \mathcal{D}_{\text{Val}} \; (16\%)\}$$
Partitioning was stratified by the binary target outcome (30-day acute care hospitalization or emergency department visit) to preserve class prevalence across all splits ($13.10\%$ in Nationwide, $14.70\%$ in Texas). 

Missing values were imputed strictly using statistics computed on the training partition $\mathcal{D}_{\text{Train}}$ to eliminate data leakage: continuous variables were imputed using median values, while categorical indicators were imputed with a missingness category token (\texttt{"Missing"}). Continuous predictors were standardized using training-derived robust scaling (subtracting median and dividing by interquartile range) for neural models and left unscaled for decision tree architectures. Categorical high-cardinality institutional identifiers were frequency-encoded using out-of-fold training counts, assigning zero frequency to novel categories encountered in validation or test partitions. Subgroup analyses were strictly restricted to demographic and clinical strata containing at least $n_{\text{pos}, g} \ge 30$ observed positive events.

\subsection{Model Development and Hyperparameter Selection}
We trained and evaluated candidate architectures across six distinct model tiers:
\begin{enumerate}
    \item \textbf{Tier 1: Classical Linear \& Tree Baselines}: Logistic Regression with $L_2$ regularization ($C \in \{0.01, 0.1, 1.0, 10.0\}$) and Random Forest ($n_{\text{trees}}=300$, max depth $= 12$, min samples split $= 10$).
    \item \textbf{Tier 2: Gradient Boosted Decision Trees (GBDT)}: LightGBM ($n_{\text{estimators}}=1000$, learning rate $= 0.03$, max depth $= 7$, num leaves $= 63$, feature fraction $= 0.8$, early stopping $= 50$), XGBoost ($n_{\text{estimators}}=1000$, learning rate $= 0.03$, max depth $= 6$, subsample $= 0.8$, colsample bytree $= 0.8$), and CatBoost ($n_{\text{iterations}}=1200$, learning rate $= 0.04$, depth $= 6$, $L_2$ leaf reg $= 3.0$).
    \item \textbf{Tier 3: Standard Deep Tabular Models}: Standard Multi-Layer Perceptron (MLP; 3 dense layers of dimensions $256 \to 128 \to 64$, ReLU activation, dropout $0.2$, Adam optimizer, learning rate $1\times 10^{-3}$) and Standard Tabular Transformer (4 multi-head self-attention layers, 8 heads, embedding dim $= 64$).
    \item \textbf{Tier 4: Hierarchical Multi-Tier Attention (HIR-M3)}: Partitioned embedding projections for Micro ($d=128$), Meso ($d=64$), and Macro ($d=32$) tiers; cross-attention module setting Query as Micro tokens and Key/Value as concatenated Meso + Macro tokens; structural attention regularization weight $\lambda_{\text{HIR}} = 0.05$.
    \item \textbf{Tier 5: Algorithmic Fairness Models (ACT-Parity v2)}: Residual clinical gating mechanism ($\mathbf{h}_{\text{final}} = \mathbf{h}_{\text{micro}} + \alpha(\mathbf{g} \odot \mathbf{h}_{\text{context}})$) coupled with Augmented Lagrangian demographic parity loss ($\mu=0.50$, tolerance $\delta=0.04$) and demographic invariance penalty ($\lambda_{\text{inv}}=0.10$).
    \item \textbf{Tier 6: Tabular Foundation Models (TFMs)}: Prior-data fitted network TabPFN (zero-shot Bayesian inference with stratified context subsampling $N \le 1024, D \le 100$), in-context meta-learner TabICL (batched inference with support prototype subsampling $N \le 1024$), and feature-tokenized TabFM backbone.
\end{enumerate}
Decision thresholds $\tau_{\text{val}}^*$ were tuned exclusively on $\mathcal{D}_{\text{Val}}$ to maximize the macro-F1 score and then evaluated without modification on $\mathcal{D}_{\text{Test}}$. Post-hoc Platt calibration slopes and Brier scores were computed across all test sets.

\subsection{Performance and Fairness Metrics}
Discriminative utility was assessed via the Area Under the Receiver Operating Characteristic Curve (AUROC), Area Under the Precision-Recall Curve (PR-AUC), balanced F1-score at threshold $\tau^*$, Sensitivity (Recall), Specificity, and Brier calibration score. Algorithmic fairness and demographic equity across 6 racial/ethnic demographic groups were quantified using:
\begin{enumerate}
    \item \textbf{Worst-Group False Negative Rate ($\text{FNR}_{\text{worst}}$)}:
    $$\text{FNR}_{\text{worst}} = \max_{g \in \mathcal{G}} \text{FNR}_g = \max_{g \in \mathcal{G}} \frac{\text{FN}_g}{\text{FN}_g + \text{TP}_g}$$
    \item \textbf{False Negative Rate Disparity Gap ($\Delta\text{FNR}$)}:
    $$\Delta\text{FNR} = \max_{g \in \mathcal{G}} \text{FNR}_g - \min_{g \in \mathcal{G}} \text{FNR}_g$$
    \item \textbf{Equalized Odds Difference ($\text{EOD}$)}:
    $$\text{EOD} = \max \left( \max_{g,g'} |\text{FNR}_g - \text{FNR}_{g'}|, \; \max_{g,g'} |\text{FPR}_g - \text{FPR}_{g'}| \right)$$
    \item \textbf{Harm-Weighted Equitable False Negative Harm Index ($\text{EFNHI}^*$)}:
    $$\text{EFNHI}^* = \sum_{g=1}^{|\mathcal{G}|} w_g \cdot \text{FNR}_g^2 \cdot (1 - \text{ROC-AUC}_g)$$
\end{enumerate}
Statistical uncertainty was quantified via $1{,}000$ stratified bootstrap iterations on the pristine test partition $\mathcal{D}_{\text{Test}}$ ($N=13{,}572$ Nationwide, $N=12{,}490$ Texas), reporting empirical $2.5\text{th}$ and $97.5\text{th}$ percentiles as 95\% confidence intervals.

\section{Exploratory and Development Analyses}

\noindent\textbf{Evaluation Protocol Reconciliation:} 
To maintain strict methodological integrity and prevent confounding between distinct experimental stages, we differentiate three evaluation regimes:
\begin{enumerate}
    \item \textbf{Protocol A (Primary Natural-Prevalence Held-Out Test Set, Tables S\ref{tab:s_all_model_performance}--\ref{tab:s_fairness})}: Evaluated strictly on the pristine 20\% test partition ($\mathcal{D}_{\text{Test}}$, $N=13{,}572$ Nationwide, $N=12{,}490$ Texas) at real-world prevalence ($13.10\%$ and $14.70\%$) using validation-tuned decision thresholds ($\tau^*$). Standalone HIR-M3 achieves $\text{AUROC} = 0.711$, $\text{F1} = 0.372$, and $\text{Brier} = 0.211$.
    \item \textbf{Protocol B (Development-Set Factorial Ablation, Table S\ref{tab:s_hir_ablation})}: Evaluated on development folds ($\mathcal{D}_{\text{Dev}}$) under balanced sampling to evaluate neural inductive biases (HIR-1 $\to$ HIR-5), where HIR-5 reaches $\text{AUROC} = 0.777$, $\text{F1} = 0.429$, and $\text{Brier} = 0.155$.
    \item \textbf{Protocol C (Convex Ensembles \& Fairness Hybrids, Table S\ref{tab:s_ensemble_sweep})}: Evaluates dual-model probability blends ($70\%$ XGBoost : $30\%$ HIR-M3) and Augmented Lagrangian fairness regularizers (ACT-Parity v2).
\end{enumerate}
